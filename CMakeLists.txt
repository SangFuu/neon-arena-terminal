cmake_minimum_required(VERSION 3.16)
project(NeonArena VERSION 2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(neon_core src/game.c src/input.c src/render.c)
target_include_directories(neon_core PUBLIC src)
if(NOT MSVC)
 target_link_libraries(neon_core PUBLIC m)
 target_compile_options(neon_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(neon_arena src/main.c)
target_link_libraries(neon_arena PRIVATE neon_core)
if(NOT MSVC)
 target_compile_options(neon_arena PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(test_game tests/test_game.c)
 target_link_libraries(test_game PRIVATE neon_core)
 add_executable(test_input tests/test_input.c)
 target_link_libraries(test_input PRIVATE neon_core)
 add_test(NAME gameplay COMMAND test_game)
 add_test(NAME clavier-terminal COMMAND test_input)
 add_test(NAME terminal COMMAND neon_arena --smoke-test 240)
 set_tests_properties(gameplay clavier-terminal terminal PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING AND WIN32)
 add_executable(test_console tests/test_console.c)
 target_link_libraries(test_console PRIVATE user32)
 add_test(NAME console-windows COMMAND test_console $<TARGET_FILE:neon_arena>)
 set_tests_properties(console-windows PROPERTIES TIMEOUT 15)
endif()
